% ---- Generar fecha corta automática ----
\newcommand{\setfechacorta}{%
  \ifnum\month<2
    \edef\@fechacorta{\the\numexpr\year-1\relax.1}% enero -> semestre 2 del año anterior
  \else
    \ifnum\month<8
      \edef\@fechacorta{\the\numexpr\year-1.2}% feb-jul -> semestre 2
    \else
      \edef\@fechacorta{\the\year.1}% ago-ene -> semestre 1
    \fi
  \fi
}
\setfechacorta

%% Portada
\newcommand{\portada}{
    \newgeometry{margin=12mm}
    \thispagestyle{empty}
    \noindent
    \fcolorbox{celeste}{celeste}{
    \begin{minipage}[t][10cm][t]{18.1cm}
    \flushleft
    \hspace{0pt}\\
    {\fontsize{45pt}{45pt}\selectfont \textbf{\@asignatura} \\}
    \vspace{\fill}
    {\fontsize{35pt}{35pt}\selectfont \@titulo \\[1cm]\hspace{0pt}}
    \end{minipage}
    }\\[9mm]
    \includegraphics[width=6cm]{figs/portada.png}
    \hspace{7mm}
    \fcolorbox{gris}{gris}{
    \begin{minipage}[b][15.59cm][t]{11.2cm}
    \flushleft
    \hspace{0pt}\\
    \Huge
    \textbf{\@autor}\\[1.5cm]
    \LARGE
    \@fecha
    \end{minipage}
    }
    
    \clearpage
    \restoregeometry
    
    \noindent
    \parbox{\linewidth}{\Huge\flushleft
    {\bfseries \@titulo}
    \ifthenelse{\equal{\@subtitulo}{}}{}{\\ \huge\@subtitulo}\\[1cm]
    }
}
